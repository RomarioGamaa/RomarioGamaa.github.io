// Validação do ciclo CRUD de reservas da GamaTravel
@Test
public void testCreateAndValidateBooking() {
    Booking newBooking = new Booking("Romário Gama", "Tóquio", "CONFIRMADO");

    given()
        .contentType("application/json")
        .body(newBooking)
    .when()
        .post("/api/v1/bookings")
    .then()
        .statusCode(201)
        .body("passengerName", equalTo("Romário Gama"))
        .body("status", equalTo("CONFIRMADO"));
}
